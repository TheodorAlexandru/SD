package com.sd.memberservice.services

import com.sd.memberservice.interfaces.IMemberService
import com.sd.memberservice.pojo.Member
import org.springframework.stereotype.Service
import java.util.concurrent.ConcurrentHashMap

@Service
class MemberService : IMemberService {

    private val members = ConcurrentHashMap<Int, Member>()

    init {
        members[1] = Member(1, "mama", "1234", "Maria Popescu")
        members[2] = Member(2, "tata", "1234", "Ion Popescu")
        members[3] = Member(3, "copil", "1234", "Andrei Popescu")
    }

    override fun createMember(member: Member) {
        members[member.id] = member
    }

    override fun getMember(id: Int): Member? {
        return members[id]
    }

    override fun getAllMembers(): List<Member> {
        return members.values.toList()
    }

    override fun updateMember(id: Int, member: Member) {
        members[id] = member
    }

    override fun deleteMember(id: Int) {
        members.remove(id)
    }

    override fun login(username: String, password: String): Member? {
        return members.values.find {
            it.username == username && it.password == password
        }
    }
}

package com.sd.memberservice.controllers

import com.sd.memberservice.interfaces.IMemberService
import com.sd.memberservice.pojo.Member
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.http.HttpStatus
import org.springframework.http.ResponseEntity
import org.springframework.web.bind.annotation.*

@RestController
class MemberController {

    @Autowired
    private lateinit var memberService: IMemberService

    @PostMapping("/members")
    fun createMember(@RequestBody member: Member): ResponseEntity<Unit> {
        memberService.createMember(member)
        return ResponseEntity(Unit, HttpStatus.CREATED)
    }

    @GetMapping("/members/{id}")
    fun getMember(@PathVariable id: Int): ResponseEntity<Member?> {
        val member = memberService.getMember(id)

        return if (member != null) {
            ResponseEntity(member, HttpStatus.OK)
        } else {
            ResponseEntity(null, HttpStatus.NOT_FOUND)
        }
    }

    @GetMapping("/members")
    fun getAllMembers(): ResponseEntity<List<Member>> {
        val members = memberService.getAllMembers()

        return if (members.isEmpty()) {
            ResponseEntity(members, HttpStatus.NO_CONTENT)
        } else {
            ResponseEntity(members, HttpStatus.OK)
        }
    }

    @PutMapping("/members/{id}")
    fun updateMember(
        @PathVariable id: Int,
        @RequestBody member: Member
    ): ResponseEntity<Unit> {
        return if (memberService.getMember(id) != null) {
            memberService.updateMember(id, member)
            ResponseEntity(Unit, HttpStatus.ACCEPTED)
        } else {
            ResponseEntity(Unit, HttpStatus.NOT_FOUND)
        }
    }

    @DeleteMapping("/members/{id}")
    fun deleteMember(@PathVariable id: Int): ResponseEntity<Unit> {
        return if (memberService.getMember(id) != null) {
            memberService.deleteMember(id)
            ResponseEntity(Unit, HttpStatus.OK)
        } else {
            ResponseEntity(Unit, HttpStatus.NOT_FOUND)
        }
    }

    @PostMapping("/login")
    fun login(@RequestBody member: Member): ResponseEntity<Member?> {
        val loggedMember = memberService.login(member.username, member.password)

        return if (loggedMember != null) {
            ResponseEntity(loggedMember, HttpStatus.OK)
        } else {
            ResponseEntity(null, HttpStatus.UNAUTHORIZED)
        }
    }
}

package com.sd.expenseservice.services

import com.sd.expenseservice.interfaces.IExpenseService
import com.sd.expenseservice.pojo.Expense
import com.sd.expenseservice.pojo.ExpenseCategory
import org.springframework.stereotype.Service
import java.util.concurrent.ConcurrentHashMap

@Service
class ExpenseService : IExpenseService {

    private val expenses = ConcurrentHashMap<Int, Expense>()

    init {
        expenses[1] = Expense(
            1,
            1,
            ExpenseCategory.MANCARE,
            "Cumparaturi supermarket",
            250.5
        )

        expenses[2] = Expense(
            2,
            2,
            ExpenseCategory.INTRETINERE,
            "Factura intretinere",
            430.0
        )

        expenses[3] = Expense(
            3,
            3,
            ExpenseCategory.SCOALA,
            "Caiete si rechizite",
            120.0
        )
    }

    override fun createExpense(expense: Expense) {
        expenses[expense.id] = expense
    }

    override fun getExpense(id: Int): Expense? {
        return expenses[id]
    }

    override fun getAllExpenses(): List<Expense> {
        return expenses.values.toList()
    }

    override fun updateExpense(id: Int, expense: Expense) {
        expenses[id] = expense
    }

    override fun deleteExpense(id: Int) {
        expenses.remove(id)
    }

    override fun searchExpenses(
        memberId: Int?,
        category: ExpenseCategory?
    ): List<Expense> {
        return expenses.values.filter {
            (memberId == null || it.memberId == memberId) &&
            (category == null || it.category == category)
        }.toList()
    }
}

package com.sd.reportservice.services

import com.sd.reportservice.pojo.Expense
import com.sd.reportservice.pojo.ExpenseCategory
import org.springframework.beans.factory.annotation.Autowired
import org.springframework.beans.factory.annotation.Value
import org.springframework.stereotype.Service
import org.springframework.web.client.RestTemplate

@Service
class ReportService {

    @Autowired
    private lateinit var restTemplate: RestTemplate

    @Value("\${expense.service.url}")
    private lateinit var expenseServiceUrl: String

    fun getAllExpenses(): List<Expense> {
        val response = restTemplate.getForObject(
            expenseServiceUrl,
            Array<Expense>::class.java
        )

        return response?.toList() ?: emptyList()
    }

    fun getTotalFamilyExpenses(): Double {
        return getAllExpenses().sumOf { it.amount }
    }

    fun getTotalForMember(memberId: Int): Double {
        return getAllExpenses()
            .filter { it.memberId == memberId }
            .sumOf { it.amount }
    }

    fun getTotalForCategory(category: ExpenseCategory): Double {
        return getAllExpenses()
            .filter { it.category == category }
            .sumOf { it.amount }
    }
}
